      function ESMF_TimeLT(time1, time2)

! !RETURN VALUE:
      logical :: ESMF_TimeLT

      type(ESMF_Time), intent(in) :: time1
      type(ESMF_Time), intent(in) :: time2
!
! !DESCRIPTION:
!     This method overloads the (<) operator for the {\tt ESMF\_Time} class.

      integer :: localrc

      ! Initialize output value in case of error
